    private void TriggerAutoSave()
    {
        _autoSaveCts?.Cancel();
        _autoSaveCts = new System.Threading.CancellationTokenSource();
        var token = _autoSaveCts.Token;
        _ = Task.Run(async () =>
        {
            await Task.Delay(800, token);
            if (!token.IsCancellationRequested)
                await InvokeAsync(AutoSaveAsync);
        }, token);
    }

    private async Task AutoSaveAsync()
    {
        if (_currentUserId == 0) return;
        if (!editId.HasValue)
        {
            bool isLainnya = model.Jenis == "Lainnya";
            if (!isLainnya && (model.PjId == 0 || model.MenyerahkanId == 0 || model.MengetahuiId == 0)) return;
            if (isLainnya && model.MenyerahkanId == 0) return;
            await CreateInitialDraftAsync();
            if (!editId.HasValue) return;
        }

        autoSaveStatus = AutoSaveState.Saving;
        StateHasChanged();
        try
        {
            if (!editId.HasValue) { autoSaveStatus = AutoSaveState.Error; StateHasChanged(); return; }
            DbContext.ChangeTracker.Clear();
            var existing = await DbContext.Set<BeritaAcaraEntity>()
                .Include(b => b.Perangkat)
                .FirstOrDefaultAsync(b => b.Id == editId.Value);
            if (existing is null) { autoSaveStatus = AutoSaveState.Error; StateHasChanged(); return; }

            existing.Tanggal = DateOnly.FromDateTime(model.Tanggal);
            existing.Jenis = model.Jenis;
            existing.JenisCustom = model.Jenis == "Lainnya" ? model.JenisCustom : null;
            existing.PjId = model.PjId;
            existing.PjNoTelp = model.PjNoTelp;
            existing.PengunaAlihDaya = model.PengunaAlihDaya;
            existing.MenyerahkanId = model.MenyerahkanId;
            existing.MengetahuiId = model.MengetahuiId;
            existing.TiketSscNo = model.TiketSscNo;
            existing.TanggalKembali = model.TanggalKembali.HasValue
                ? DateOnly.FromDateTime(model.TanggalKembali.Value) : null;

            var validPerangkat = model.Perangkat.Where(p => p.BarangId > 0).ToList();
            if (validPerangkat.Any())
            {
                DbContext.PerangkatBA.RemoveRange(existing.Perangkat.ToList());
                existing.Perangkat = validPerangkat.Select(p => new PerangkatBA
                {
                    BaId = existing.Id,
                    BarangId = p.BarangId,
                    Jumlah = p.Jumlah > 0 ? p.Jumlah : 1,
                    Satuan = string.IsNullOrWhiteSpace(p.Satuan) ? "Pcs" : p.Satuan,
                    NoSerial = p.NoSerial,
                    Keterangan = p.Keterangan
                }).ToList();
            }

            await DbContext.SaveChangesAsync();
            autoSaveStatus = AutoSaveState.Saved;
            StateHasChanged();
            await Task.Delay(3000);
            if (autoSaveStatus == AutoSaveState.Saved) { autoSaveStatus = AutoSaveState.Idle; StateHasChanged(); }
        }
        catch (Exception ex)
        {
            Logger.LogError(ex, "[AutoSaveAsync] Error");
            autoSaveStatus = AutoSaveState.Error;
            StateHasChanged();
        }
    }

                StateHasChanged();
            }
        }
        else
        {
            foreach (var file in newFiles)

                photoFiles.Add(file);

                {
                    using var stream = file.OpenReadStream(5_000_000);
                    using var ms = new MemoryStream();

    private void RemoveNewPhoto(int index)
    {
        if (index >= 0 && index < photoDataUrls.Count)

            photoDataUrls.RemoveAt(index);
            if (index < photoFiles.Count) photoFiles.RemoveAt(index);
        }
    }

    private async Task RemoveExistingPhoto(int index)
    {
        if (index < 0 || index >= existingPhotoUrls.Count) return;
        var (relPath, _) = existingPhotoUrls[index];
        existingPhotoUrls.RemoveAt(index);
        photosChanged = true;
        if (editId.HasValue)
        {
            try
            {
                var foto = await DbContext.BuktiFoto.FirstOrDefaultAsync(f => f.BaId == editId.Value && f.FilePath == relPath);
                if (foto != null)
                {
                    DbContext.BuktiFoto.Remove(foto);
                    await DbContext.SaveChangesAsync();
                    var physPath = Path.Combine(Directory.GetCurrentDirectory(), "wwwroot", relPath.TrimStart('/').Replace('/', Path.DirectorySeparatorChar));
                    if (File.Exists(physPath)) File.Delete(physPath);
                }
            }
            catch (Exception ex) { Logger.LogError(ex, "RemoveExistingPhoto error"); }
        }
    }

    private async Task SubmitAsync()
    {
        errorMessage = string.Empty;

        if (model.Jenis == "Lainnya")
        {
            if (string.IsNullOrWhiteSpace(model.JenisCustom)) { errorMessage = "Nama/Jenis Berita Acara wajib diisi."; return; }
            await SubmitLainnyaAsync();
            return;
        }

        if (model.PjId == 0) { errorMessage = "Pilih Penanggung Jawab (PJ) terlebih dahulu."; return; }
        if (model.TipePj == "TenagaAlihDaya" && string.IsNullOrWhiteSpace(model.PengunaAlihDaya)) { errorMessage = "Nama Pengguna TAD wajib diisi."; return; }
        if (model.MenyerahkanId == 0 || !adminGudangUsers.Any(u => u.PegawaiId == model.MenyerahkanId)) { errorMessage = "Pilih Admin (Yang Menyerahkan) yang valid."; return; }
        if (model.MengetahuiId == 0 || !approvers.Any(u => u.Id == model.MengetahuiId)) { errorMessage = "Pilih Approver (Yang Mengetahui) yang valid."; return; }
        if (model.Jenis == "Peminjaman" && !model.TanggalKembali.HasValue) { errorMessage = "Tanggal Kembali wajib diisi untuk Peminjaman."; return; }
        if (!model.Perangkat.Any(p => p.BarangId > 0)) { errorMessage = "Tambahkan minimal satu perangkat."; return; }
        if (model.Jenis == "Peminjaman" && model.TanggalKembali.HasValue && model.TanggalKembali.Value.Date < model.Tanggal.Date)
        { errorMessage = "Tanggal Kembali tidak boleh sebelum Tanggal peminjaman."; return; }

        await JSRuntime.InvokeVoidAsync("appLoading.show", "Menyiapkan dokumen...", "Sedang memproses, harap tunggu");
        isSaving = true;
        try
        {
            if (_currentUserId == 0)
            {
                var auth = await AuthStateProvider.GetAuthenticationStateAsync();
                var userIdStr = auth.User.FindFirst(System.Security.Claims.ClaimTypes.NameIdentifier)?.Value;
                int.TryParse(userIdStr, out _currentUserId);
            }
            if (_currentUserId == 0) { errorMessage = "Sesi habis, silakan login kembali."; await JSRuntime.InvokeVoidAsync("appLoading.hide"); isSaving = false; return; }

            if (!editId.HasValue) await CreateInitialDraftAsync();

            if (editId.HasValue)
            {
                DbContext.ChangeTracker.Clear();
                var existing = await DbContext.Set<BeritaAcaraEntity>()
                    .Include(b => b.Perangkat)
                    .Include(b => b.BuktiFotos)
                    .FirstOrDefaultAsync(b => b.Id == editId.Value);

                if (existing is null) { errorMessage = "Draft tidak ditemukan."; isSaving = false; return; }

                existing.DocxPath = null; // force regenerate
                existing.Tanggal = DateOnly.FromDateTime(model.Tanggal);
                existing.Jenis = model.Jenis;
                existing.JenisCustom = null;
                existing.PjId = model.PjId;
                existing.PjNoTelp = model.PjNoTelp;
                existing.PengunaAlihDaya = model.PengunaAlihDaya;
                existing.MenyerahkanId = model.MenyerahkanId;
                existing.MengetahuiId = model.MengetahuiId;
                existing.TiketSscNo = model.TiketSscNo;
                existing.TanggalKembali = model.TanggalKembali.HasValue
                    ? DateOnly.FromDateTime(model.TanggalKembali.Value) : null;

                if (existing.Status == "Rejected")
                {
                    existing.Status = "Draft";
                    existing.AlasanReject = null;
                    existing.TtdPjPath = null;
                    existing.SubmittedAt = null;
                    existing.PjSignedAt = null;
                    existing.ApprovedAt = null;
                }

                DbContext.PerangkatBA.RemoveRange(existing.Perangkat.ToList());
                existing.Perangkat = model.Perangkat
                    .Where(p => p.BarangId > 0 && p.Jumlah > 0)
                    .Select(p => new PerangkatBA
                    {
                        BaId = existing.Id,
                        BarangId = p.BarangId,
                        Jumlah = p.Jumlah,
                        Satuan = string.IsNullOrWhiteSpace(p.Satuan) ? "Pcs" : p.Satuan,
                        NoSerial = p.NoSerial,
                        Keterangan = p.Keterangan
                    }).ToList();

                if (photoFiles.Any() && !existingPhotoUrls.Any())
                {
                    var photoFolder = Path.Combine(Directory.GetCurrentDirectory(), "wwwroot", "files", "photos");
                    Directory.CreateDirectory(photoFolder);
                    int idx2 = 1;
                    foreach (var file in photoFiles)
                    {
                        var ext = Path.GetExtension(file.Name);
                        var saveName = "ba--photo--1782787019862";
                        await using var fs = File.Create(Path.Combine(photoFolder, saveName));
                        await file.OpenReadStream(5_000_000).CopyToAsync(fs);
                        DbContext.BuktiFoto.Add(new BuktiFoto { BaId = existing.Id, BeritaAcara = existing, FilePath = "files/photos/" });
                        idx2++;
                    }
                }

                await DbContext.SaveChangesAsync();
                Navigation.NavigateTo("/editor/");
            }
            else
            {
                var ba = new BeritaAcaraEntity
                {
                    Tanggal = DateOnly.FromDateTime(model.Tanggal),
                    Jenis = model.Jenis,
                    PjId = model.PjId,
                    PjNoTelp = model.PjNoTelp,
                    PengunaAlihDaya = model.PengunaAlihDaya,
                    MenyerahkanId = model.MenyerahkanId,
                    MengetahuiId = model.MengetahuiId,
                    TiketSscNo = model.TiketSscNo,
                    TanggalKembali = model.TanggalKembali.HasValue ? DateOnly.FromDateTime(model.TanggalKembali.Value) : null,
                    Status = "Draft",
                    CreatedBy = _currentUserId,
                    CreatedAt = DateTime.Now,
                    Perangkat = model.Perangkat
                        .Where(p => p.BarangId > 0 && p.Jumlah > 0)
                        .Select(p => new PerangkatBA
                        {
                            BarangId = p.BarangId,
                            Jumlah = p.Jumlah,
                            Satuan = string.IsNullOrWhiteSpace(p.Satuan) ? "Pcs" : p.Satuan,
                            NoSerial = p.NoSerial,
                            Keterangan = p.Keterangan

                DbContext.Set<BeritaAcaraEntity>().Add(ba);
                await DbContext.SaveChangesAsync();
                Navigation.NavigateTo("/editor/");

        catch (Exception ex)
        {
            var inner = ex.InnerException != null ? "\nInner: " : "";
            errorMessage = "Gagal menyimpan Berita Acara: ";
            await JSRuntime.InvokeVoidAsync("appLoading.hide");
        }
        finally { isSaving = false; }
    }

    private async Task SubmitLainnyaAsync()
    {
        await JSRuntime.InvokeVoidAsync("appLoading.show", "Menyimpan ke Arsip...", "Sedang memproses");
        isSaving = true;
        try
        {
            if (_currentUserId == 0)
            {
                var auth = await AuthStateProvider.GetAuthenticationStateAsync();
                var userIdStr = auth.User.FindFirst(System.Security.Claims.ClaimTypes.NameIdentifier)?.Value;
                int.TryParse(userIdStr, out _currentUserId);
            }

            if (currentUserObj == null)
                currentUserObj = adminGudangUsers.FirstOrDefault(u => u.Id == _currentUserId);

            if (!editId.HasValue) await CreateInitialDraftAsync();

            if (!editId.HasValue)
            {
                var baNew = new BeritaAcaraEntity
                {
                    Tanggal = DateOnly.FromDateTime(model.Tanggal),
                    Jenis = "Lainnya",
                    JenisCustom = model.JenisCustom,
                    MenyerahkanId = model.MenyerahkanId,
                    MengetahuiId = 0,
                    PjId = 0,
                    Status = "Draft",
                    CreatedBy = _currentUserId,
                    CreatedAt = DateTime.Now,
                };
                DbContext.Set<BeritaAcaraEntity>().Add(baNew);
                await DbContext.SaveChangesAsync();
                editId = baNew.Id;
            }

            DbContext.ChangeTracker.Clear();
            var ba = await DbContext.Set<BeritaAcaraEntity>().FindAsync(editId.Value);
            if (ba is null) { errorMessage = "Gagal menemukan draft."; isSaving = false; await JSRuntime.InvokeVoidAsync("appLoading.hide"); return; }

            var (counterValue, nomorSurat) = await BACounterService.GetNextNomorSuratAsync(DateOnly.FromDateTime(model.Tanggal));
            ba.NomorSurat = nomorSurat;
            ba.CounterValue = counterValue;
            ba.Jenis = "Lainnya";
            ba.JenisCustom = model.JenisCustom;
            ba.Status = "Approved";
            ba.ApprovedAt = DateTime.Now;
            ba.SubmittedAt = DateTime.Now;

            ba.DocxPath = await DocumentService.GenerateLainnyaDocxAsync(ba);
            await DbContext.SaveChangesAsync();
            Navigation.NavigateTo("/archive");
        }
        catch (Exception ex)
        {
            var inner = ex.InnerException != null ? "\nInner: " : "";
            errorMessage = "Gagal menyimpan: ";
            await JSRuntime.InvokeVoidAsync("appLoading.hide");
        }
        finally { isSaving = false; }
    }

    private void OpenMengetahuiSearch()
    {
        mengetahuiSearchQuery = string.Empty;
        mengetahuiSearchResults = approvers.Take(8).ToList();
        showMengetahuiSearch = true;
    }

    private void CloseMengetahuiSearch() => showMengetahuiSearch = false;

    private void OnMengetahuiSearchInput(ChangeEventArgs e)
    {
        mengetahuiSearchQuery = e.Value?.ToString() ?? string.Empty;
        mengetahuiSearchResults = string.IsNullOrWhiteSpace(mengetahuiSearchQuery)
            ? approvers.Take(8).ToList()
            : approvers.Where(u => u.Nama?.Contains(mengetahuiSearchQuery, StringComparison.OrdinalIgnoreCase) ?? false).Take(8).ToList();
    }

    private void SelectMengetahui(ApplicationUser u)
    {
        model.MengetahuiId = u.Id;
        showMengetahuiSearch = false;
        _ = AutoSaveAsync();
    }
